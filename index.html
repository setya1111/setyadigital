/* ============================================================ RESOURCE SISWA ============================================================ */ function renderLessonResources(resources) { const container = document.getElementById( 'currentLessonResources' ); if (!container) return; if ( !Array.isArray(resources) || resources.length === 0 ) { container.innerHTML = `
Tidak ada resource untuk materi ini.
`; return; } container.innerHTML = resources .filter( resource => resource && resource.url && String(resource.url).trim() !== '' ) .map((resource, index) => { const title = resource.title || `Resource ${index + 1}`; /* * Pastikan URL memiliki protocol. * Kalau admin memasukkan: * gemini.google.com * * akan otomatis menjadi: * https://gemini.google.com */ let url = String(resource.url).trim(); if ( !/^https?:\/\//i.test(url) ) { url = 'https://' + url; } /* * Escape untuk keamanan HTML. */ const safeTitle = esc(title); const safeUrl = esc(url); return `
${safeTitle}
${safeUrl}
Buka
`; }) .join(''); } /* ============================================================ COPY RESOURCE LINK ============================================================ */ async function copyResourceLink(url) { if (!url) { toast( 'Link resource kosong.', 'error' ); return; } try { await navigator.clipboard.writeText( url ); toast( 'Link berhasil disalin.', 'success' ); } catch (error) { /* * Fallback untuk browser yang * tidak mengizinkan clipboard API. */ const textarea = document.createElement( 'textarea' ); textarea.value = url; textarea.style.position = 'fixed'; textarea.style.opacity = '0'; document.body.appendChild( textarea ); textarea.focus(); textarea.select(); try { document.execCommand( 'copy' ); toast( 'Link berhasil disalin.', 'success' ); } catch (err) { toast( 'Gagal menyalin link.', 'error' ); } document.body.removeChild( textarea ); } } /* ============================================================ OPEN RESOURCE ============================================================ */ /* * Fungsi ini sengaja tetap tersedia * untuk kompatibilitas dengan kode lama. * * Tombol Buka yang baru TIDAK menggunakan * fungsi ini lagi. Tombol Buka langsung * menggunakan href. */ function openResourceLink(url) { if (!url) { toast( 'Link resource kosong.', 'error' ); return; } let targetUrl = String(url).trim(); if ( !/^https?:\/\//i.test(targetUrl) ) { targetUrl = 'https://' + targetUrl; } window.location.href = targetUrl; }
